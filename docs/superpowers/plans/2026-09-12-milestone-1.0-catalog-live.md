```ts
/** Integer paise → "₹14,999" (en-IN grouping, no decimals). */
export function formatInr(paise: number): string {
  const rupees = Math.round(paise / 100);
  return '₹' + rupees.toLocaleString('en-IN', { maximumFractionDigits: 0 });
}
/** Seed content is written in rupees for readability; the DB stores paise. */
export function rupeesToPaise(rupees: number): number {
  if (!Number.isInteger(rupees) || rupees < 0) throw new Error(`rupees must be a non-negative integer, got ${rupees}`);
  return rupees * 100;
}
```

```ts
import { z } from 'zod';
import { LIMITS, MONTH_RE, SLUG_RE, SORTS, THEMES } from '../constants';

const csv = (inner: z.ZodType<string>) =>
  z.preprocess((v) => (typeof v === 'string' ? v.split(',').filter(Boolean) : v), z.array(inner));

/** Query string of GET /packages. Arrays arrive as CSV (`?destination=goa,kerala`). */
export const searchParamsSchema = z.object({
  destination: csv(z.string().regex(SLUG_RE)).optional(),
  maxBudget: z.coerce.number().int().positive().optional(), // rupees per person
  nightsMin: z.coerce.number().int().min(1).max(30).optional(),
  nightsMax: z.coerce.number().int().min(1).max(30).optional(),
  themes: csv(z.enum(THEMES)).optional(),
  month: z.string().regex(MONTH_RE).optional(), // YYYY-MM
  sort: z.enum(SORTS).default('price-asc'),
});
export type SearchParams = z.infer<typeof searchParamsSchema>;

export const hotelSchema = z.object({ name: z.string().min(1), city: z.string().min(1), stars: z.number().int().min(1).max(5), nights: z.number().int().min(1) });
export const faqSchema = z.object({ q: z.string().min(1), a: z.string().min(1) });
export const itineraryDaySchema = z.object({
  dayNo: z.number().int().min(1),
  title: z.string().min(1),
  description: z.string().min(1),
  mealB: z.boolean(), mealL: z.boolean(), mealD: z.boolean(),
  stay: z.string().nullable(),
  locationName: z.string().nullable().optional(),
  lat: z.number().nullable().optional(),
  lng: z.number().nullable().optional(),
});
export const departureSchema = z.object({
  date: z.string().regex(/^\d{4}-\d{2}-\d{2}$/),
  seatsTotal: z.number().int().min(0),
  guaranteed: z.boolean().default(false),
  priceDoublePaise: z.number().int().min(0),
  priceTriplePaise: z.number().int().min(0),
  priceChildPaise: z.number().int().min(0),
  singleSupplementPaise: z.number().int().min(0),
});
export const packageSchema = z
  .object({
    slug: z.string().regex(SLUG_RE),
    destinationSlug: z.string().regex(SLUG_RE),
    name: z.string().min(1).max(80),
    summary: z.string().min(1).max(400),
    themes: z.array(z.enum(THEMES)).min(1).max(LIMITS.maxThemesPerPackage),
    nights: z.number().int().min(1).max(30),
```

```ts
import { logger } from 'hono/logger';

import { health } from './routes/public/health';

export function createApp() {
  const app = new OpenAPIHono({
    defaultHook: (result, c) => {
      if (!result.success) {
        const fieldErrors: Record<string, string> = {};
        for (const i of result.error.issues) fieldErrors[i.path.join('.') || '_'] = i.message;
        return c.json(new ApiError('validation', 'Invalid request', fieldErrors).body(), 400);
      }
    },
  });
  app.use('*', logger());
  app.onError((err, c) => {
    if (err instanceof ApiError) return c.json(err.body(), err.status);
    console.error(err);
    return c.json(new ApiError('internal', 'Something went wrong').body(), 500);
  });
  app.notFound((c) => c.json(new ApiError('not_found', `No route for ${c.req.method} ${c.req.path}`).body(), 404));
  app.route('/', health);
  app.doc('/openapi.json', { openapi: '3.1.0', info: { title: 'Tripsmith API', version: '1.0.0', description: 'Public catalog API for tripsmith.vercel.app' } });
```

```ts
import type { ApiError } from '@tripsmith/shared';
const BASE = process.env.API_URL ?? 'http://localhost:8787';
export class ApiRequestError extends Error {
  constructor(public status: number, public body: ApiError['error']) { super(body.message); }
}
/** Server-side typed fetch to the Hono api. Tags feed on-demand revalidation. */
export async function api<T>(path: string, init: { tags?: string[]; revalidate?: number | false; searchParams?: Record<string, string | undefined> } = {}): Promise<T> {
  const url = new URL(path, BASE);
  for (const [k, v] of Object.entries(init.searchParams ?? {})) if (v) url.searchParams.set(k, v);
  const res = await fetch(url, { next: { tags: init.tags, revalidate: init.revalidate } });
  if (!res.ok) {
    const body = (await res.json().catch(() => ({ error: { code: 'internal', message: res.statusText } }))) as ApiError;
    throw new ApiRequestError(res.status, body.error);
  }
  return res.json() as Promise<T>;
}
```

```ts
import { LIMITS, type Badge } from '@tripsmith/shared';

export function badgeFor(d: { seatsLeft: number; guaranteed: boolean }): Badge | null {
  if (d.seatsLeft <= 0) return 'sold-out';
  if (d.seatsLeft <= LIMITS.fillingFastSeats) return 'filling-fast';
  if (d.guaranteed) return 'guaranteed';
  return null;
}
/** ISO date string for `today` in UTC (departure dates are calendar dates). */
export function isoDay(d: Date): string { return d.toISOString().slice(0, 10); }
export function isBookable(d: { date: string; seatsLeft: number }, today: Date): boolean {
  return d.seatsLeft > 0 && d.date >= isoDay(today);
}
export function startingPricePaise(departures: { date: string; seatsLeft: number; priceDoublePaise: number }[], today: Date): number | null {
  const prices = departures.filter((d) => isBookable(d, today)).map((d) => d.priceDoublePaise);
  return prices.length ? Math.min(...prices) : null;
}
```

```ts
export type DayContent = { day: number; title: string; description: string; meals: string; stay: string | null; location?: { name: string; lat: number; lng: number } }; // meals: subset of 'BLD', '' for none

export type PackageContent = {
  slug: string; destination: string; name: string; summary: string; themes: Theme[]; nights: number; departureCity: string;
  highlights: string[]; inclusions: string[]; exclusions: string[]; hotels: HotelContent[]; faq: { q: string; a: string }[];
  featured?: boolean; itinerary: DayContent[]; departures: DepartureContent[]; images: ImageContent[];
};
export type DestinationContent = { slug: string; name: string; tagline: string; intro: string; region: string; bestMonths: number[]; position: number; cover: ImageContent };
export type TestimonialContent = { name: string; city: string; text: string; rating: number; packageSlug?: string };

export const defineDestination = (c: DestinationContent): DestinationContent => c;

export class ContentError extends Error { constructor(slug: string, detail: string) { super(`content/${slug}: ${detail}`); } }

const mealsOf = (m: string) => {
  if (!/^[BLD]*$/.test(m)) throw new Error(`meals must be a subset of "BLD", got "${m}"`);
  return { mealB: m.includes('B'), mealL: m.includes('L'), mealD: m.includes('D') };
};

/** Rupees → paise, meals string → booleans, then validated by the shared schema. */
export function toPackageInput(c: PackageContent): PackageInput {
  const raw = {
    slug: c.slug, destinationSlug: c.destination, name: c.name, summary: c.summary, themes: c.themes,
    nights: c.nights, days: c.nights + 1, departureCity: c.departureCity,
    highlights: c.highlights, inclusions: c.inclusions, exclusions: c.exclusions, hotels: c.hotels, faq: c.faq, featured: c.featured ?? false,
    itinerary: c.itinerary.map((d) => ({ dayNo: d.day, title: d.title, description: d.description, ...mealsOf(d.meals), stay: d.stay, locationName: d.location?.name ?? null, lat: d.location?.lat ?? null, lng: d.location?.lng ?? null })),
    departures: c.departures.map((d) => ({ date: d.date, seatsTotal: d.seatsTotal, guaranteed: d.guaranteed ?? false, priceDoublePaise: rupeesToPaise(d.priceDouble), priceTriplePaise: rupeesToPaise(d.priceTriple), priceChildPaise: rupeesToPaise(d.priceChild), singleSupplementPaise: rupeesToPaise(d.singleSupplement) })),
  };
  const r = packageSchema.safeParse(raw);
  if (!r.success) throw new ContentError(c.slug, r.error.issues.map((i) => `${i.path.join('.')}: ${i.message}`).join('; '));
  if (c.images.length < 1) throw new ContentError(c.slug, 'needs at least one image');
  return r.data;
}
export function toDestinationInput(c: DestinationContent): DestinationInput {
  const r = destinationSchema.safeParse({ slug: c.slug, name: c.name, tagline: c.tagline, intro: c.intro, region: c.region, bestMonths: c.bestMonths, position: c.position });
  if (!r.success) throw new ContentError(c.slug, r.error.issues.map((i) => `${i.path.join('.')}: ${i.message}`).join('; '));
  return r.data;
}
```

```ts
import { put } from '@vercel/blob';
import { imageSize } from 'image-size';
import { readFile, mkdir, copyFile } from 'node:fs/promises';
import path from 'node:path';
export type Uploaded = { url: string; width: number; height: number };
export type Uploader = (file: { path: string; key: string }) => Promise<Uploaded>;
const dims = (buf: Buffer) => { const { width, height } = imageSize(buf); if (!width || !height) throw new Error('cannot read image size'); return { width, height }; };
/** Deterministic key → re-seeding overwrites instead of piling up copies. */
export const blobUploader = (token: string): Uploader => async (f) => {
  const buf = await readFile(f.path);
  const { url } = await put(f.key, buf, { access: 'public', token, addRandomSuffix: false, allowOverwrite: true, contentType: 'image/jpeg', cacheControlMaxAge: 31536000 });
  return { url, ...dims(buf) };
};
/** Dev without a Blob token: copy into web/public and serve from the site origin. */
export const localUploader = (publicDir: string, baseUrl: string): Uploader => async (f) => {
  const dest = path.join(publicDir, f.key);
  await mkdir(path.dirname(dest), { recursive: true });
  await copyFile(f.path, dest);
  return { url: `${baseUrl}/${f.key.replace(/\\/g, '/')}`, ...dims(await readFile(f.path)) };
};
```

```ts
const slugId = (prefix: string, slug: string) => `${prefix}_${slug.replace(/[^a-z0-9]/g, '')}`;

export async function seedCatalog(db: Db, content: Content, deps: { upload: Uploader; today?: Date; log?: (m: string) => void }) {
  const today = deps.today ?? new Date(); const log = deps.log ?? (() => {});

  const destIds = new Map<string, string>();

    const input = toDestinationInput(d); const cover = await upload(d.cover.file);
    const id = slugId('dest', d.slug);
    await db.insert(s.destinations).values({ id, ...input, coverUrl: cover.url }).onConflictDoUpdate({ target: s.destinations.slug, set: { name: input.name, tagline: input.tagline, intro: input.intro, region: input.region, bestMonths: input.bestMonths, position: input.position, coverUrl: cover.url, updatedAt: new Date() } });
    const [row] = await db.select({ id: s.destinations.id }).from(s.destinations).where(eq(s.destinations.slug, d.slug));
    destIds.set(d.slug, row.id); log(`destination ${d.slug}`);
  }
  let images = 0;
  for (const p of content.packages) {
    const input = toPackageInput(p);
    const destinationId = destIds.get(p.destination); if (!destinationId) throw new Error(`${p.slug}: unknown destination ${p.destination}`);
    const base = { destinationId, name: input.name, summary: input.summary, themes: input.themes, nights: input.nights, days: input.days, departureCity: input.departureCity, highlights: input.highlights, inclusions: input.inclusions, exclusions: input.exclusions, hotels: input.hotels, faq: input.faq, featured: input.featured, status: 'live' as const };
    await db.insert(s.packages).values({ id: slugId('pkg', p.slug), slug: p.slug, ...base }).onConflictDoUpdate({ target: s.packages.slug, set: { ...base, updatedAt: new Date() } });
    const [{ id: packageId }] = await db.select({ id: s.packages.id }).from(s.packages).where(eq(s.packages.slug, p.slug));
    // children are replaced wholesale — content files are the source of truth
    await db.update(s.packages).set({ coverImageId: null }).where(eq(s.packages.id, packageId));
    await db.delete(s.itineraryDays).where(eq(s.itineraryDays.packageId, packageId));
    await db.delete(s.departures).where(eq(s.departures.packageId, packageId));
    await db.delete(s.packageImages).where(eq(s.packageImages.packageId, packageId));
    await db.insert(s.itineraryDays).values(input.itinerary.map((d) => ({ id: createId(), packageId, dayNo: d.dayNo, title: d.title, description: d.description, mealB: d.mealB, mealL: d.mealL, mealD: d.mealD, stay: d.stay, locationName: d.locationName ?? null, lat: d.lat == null ? null : String(d.lat), lng: d.lng == null ? null : String(d.lng) })));
    await db.insert(s.departures).values(input.departures.map((d) => ({ id: createId(), packageId, ...d })));
    const imgRows = [] as (typeof s.packageImages.$inferInsert)[];
    for (const [i, img] of p.images.entries()) { const u = await upload(img.file); imgRows.push({ id: createId(), packageId, url: u.url, alt: img.alt, width: u.width, height: u.height, position: i }); }
    await db.insert(s.packageImages).values(imgRows); images += imgRows.length;
    const starting = startingPricePaise(input.departures.map((d) => ({ date: d.date, seatsLeft: d.seatsTotal, priceDoublePaise: d.priceDoublePaise })), today);
    await db.update(s.packages).set({ coverImageId: imgRows[0].id, startingPricePaise: starting }).where(eq(s.packages.id, packageId));
    log(`package ${p.slug} (${imgRows.length} images, from ${starting})`);
  }
  await db.delete(s.testimonials);
  if (content.testimonials.length) {
    const pkgIds = new Map((await db.select({ id: s.packages.id, slug: s.packages.slug }).from(s.packages)).map((r) => [r.slug, r.id]));

  }
  await db.execute(sql`select 1`);
  return { destinations: content.destinations.length, packages: content.packages.length, images, testimonials: content.testimonials.length };
}
```

```ts
/** Listing-card badge: the most urgent signal across the package's future departures. */
export function cardBadge(departures: { date: string; seatsLeft: number; guaranteed: boolean }[], today: Date): Badge | null {
  const future = departures.filter((d) => d.date >= isoDay(today));
  if (!future.length) return null;
  if (future.some((d) => d.seatsLeft > 0 && d.seatsLeft <= LIMITS.fillingFastSeats)) return 'filling-fast';
  if (future.some((d) => d.seatsLeft > 0 && d.guaranteed)) return 'guaranteed';
  if (future.every((d) => d.seatsLeft <= 0)) return 'sold-out';
  return null;
}
```

```ts
/** Card = package row + its destination + cover + its (future) departures. Starting price is recomputed from departures so a stale cache column never leaks. */
export function toCard(p: PackageRow, destination: { slug: string; name: string }, cover: ImageRow | null | undefined, departures: DepartureRow[], today: Date): PackageCard {
  const future = departures.filter((d) => d.date >= isoDay(today)).sort((a, b) => a.date.localeCompare(b.date));
  const next = future.find((d) => d.seatsLeft > 0) ?? null;
  const hotel = p.hotels[0];
  return {
    slug: p.slug, name: p.name, destination, nights: p.nights, days: p.days, themes: p.themes as Theme[],
    startingPricePaise: startingPricePaise(departures, today), cover: toImage(cover), highlights: p.highlights.slice(0, 3),
    hotelStars: hotel?.stars ?? 0, hotelName: hotel?.name ?? '', badge: cardBadge(departures, today) as Badge | null, nextDeparture: next?.date ?? null,
  };
}
```

```ts
import { and, arrayOverlaps, asc, eq, exists, gte, inArray, isNotNull, lte, sql } from 'drizzle-orm';
import type { SearchParams, SearchResult } from '@tripsmith/shared';
import type { Db } from '../../infra/db/client';
import * as s from '../../infra/db/schema';
import { toCard, type DepartureRow } from './mappers';

/** Loads live packages + destination + cover + future departures (with seats_left) in three small queries. Shared by search, destinations and related. */
export async function loadCards(db: Db, where: ReturnType<typeof and> | undefined, today: Date) {
  const rows = await db.select({ p: s.packages, d: { slug: s.destinations.slug, name: s.destinations.name }, cover: s.packageImages })
    .from(s.packages).innerJoin(s.destinations, eq(s.destinations.id, s.packages.destinationId))
    .leftJoin(s.packageImages, eq(s.packageImages.id, s.packages.coverImageId))
    .where(and(eq(s.packages.status, 'live'), where));
  if (!rows.length) return [];
  const ids = rows.map((r) => r.p.id);
  const deps = await db.select({ d: s.departures, seatsLeft: s.departureAvailability.seatsLeft })
    .from(s.departures).innerJoin(s.departureAvailability, eq(s.departureAvailability.departureId, s.departures.id))
    .where(inArray(s.departures.packageId, ids));
  const byPkg = new Map<string, DepartureRow[]>();
  for (const { d, seatsLeft } of deps) byPkg.set(d.packageId, [...(byPkg.get(d.packageId) ?? []), { ...d, seatsLeft }]);
  return rows.map((r) => toCard(r.p, r.d, r.cover, byPkg.get(r.p.id) ?? [], today));
}

/** The one search implementation (R3) — reused by the v3 `searchPackages` tool. */
export async function searchPackages(db: Db, params: SearchParams, today = new Date()): Promise<SearchResult> {
  const conds = [] as ReturnType<typeof eq>[];
  if (params.destination?.length) conds.push(inArray(s.destinations.slug, params.destination));
  if (params.themes?.length) conds.push(arrayOverlaps(s.packages.themes, params.themes));
  if (params.nightsMin) conds.push(gte(s.packages.nights, params.nightsMin));
  if (params.nightsMax) conds.push(lte(s.packages.nights, params.nightsMax));
  if (params.maxBudget) conds.push(and(isNotNull(s.packages.startingPricePaise), lte(s.packages.startingPricePaise, params.maxBudget * 100))!);
  if (params.month) {
    const d = s.departures; const a = s.departureAvailability;
    conds.push(exists(db.select({ one: sql`1` }).from(d).innerJoin(a, eq(a.departureId, d.id))
      .where(and(eq(d.packageId, s.packages.id), sql`to_char(${d.date}, 'YYYY-MM') = ${params.month}`, sql`${a.seatsLeft} > 0`))));
  }
  const items = await loadCards(db, conds.length ? and(...conds) : undefined, today);
  const byPrice = (a: number | null, b: number | null) => (a ?? Infinity) - (b ?? Infinity);
  items.sort((x, y) => params.sort === 'price-desc' ? -byPrice(x.startingPricePaise, y.startingPricePaise) || x.slug.localeCompare(y.slug)
    : params.sort === 'duration' ? x.nights - y.nights || byPrice(x.startingPricePaise, y.startingPricePaise)
    : byPrice(x.startingPricePaise, y.startingPricePaise) || x.slug.localeCompare(y.slug));
  return { items, total: items.length };
}
```

```ts
export function formatDate(iso: string): string {
  const [y, m, d] = iso.split('-').map(Number);
  const dt = new Date(Date.UTC(y, m - 1, d));
```

```ts
/** Stagger-reveal children matching `selector`; renders at rest under prefers-reduced-motion. */
export function useReveal(scope: RefObject<HTMLElement | null>, selector: string, deps: unknown[] = []) {
  useGSAP(() => {
    const mm = gsap.matchMedia();
    mm.add('(prefers-reduced-motion: no-preference)', () => {
      gsap.from(selector, { y: 18, autoAlpha: 0, duration: 0.7, ease: 'power3.out', stagger: 0.06, clearProps: 'all' });
    });
    return () => mm.revert();
  }, { scope, dependencies: deps });
}
```

```ts
import { revalidateTag } from 'next/cache';
import { NextResponse } from 'next/server';
import { z } from 'zod';
const body = z.object({ secret: z.string(), tags: z.array(z.string().min(1)).min(1).max(50) });
export async function POST(req: Request) {
  const parsed = body.safeParse(await req.json().catch(() => null));
  if (!parsed.success) return NextResponse.json({ error: { code: 'validation', message: 'secret and tags[] required' } }, { status: 400 });
  if (parsed.data.secret !== process.env.REVALIDATE_SECRET) return NextResponse.json({ error: { code: 'unauthorized', message: 'bad secret' } }, { status: 401 });
```

```ts
import { searchParamsSchema, type SearchParams } from '@tripsmith/shared';
type Raw = Record<string, string | string[] | undefined>;
/** Tolerant parse: an invalid single field is dropped rather than failing the page. */
export function parseSearch(raw: Raw): SearchParams {
  const flat: Record<string, string> = {};
  for (const [k, v] of Object.entries(raw)) if (v !== undefined) flat[k] = Array.isArray(v) ? v.join(',') : v;
  const out: Record<string, unknown> = {};
  for (const [k, v] of Object.entries(flat)) {
    const one = searchParamsSchema.pick({ [k]: true } as never).safeParse({ [k]: v });
    if (one.success) Object.assign(out, one.data);
    else if (k === 'themes' || k === 'destination') { // keep the valid members of a CSV list
      const kept = v.split(',').filter((m) => searchParamsSchema.pick({ [k]: true } as never).safeParse({ [k]: m }).success);
      if (kept.length) out[k] = kept;
    }
  }
  return searchParamsSchema.parse(out);
}
export function toQuery(p: Partial<SearchParams>): string {
  const q = new URLSearchParams();
  for (const k of ['destination', 'maxBudget', 'month', 'nightsMax', 'nightsMin', 'themes'] as const) {
    const v = p[k]; if (v === undefined) continue;
    q.set(k, Array.isArray(v) ? v.join(',') : String(v));
  }
  if (p.sort && p.sort !== 'price-asc') q.set('sort', p.sort);
  return q.toString();
}
export const withParam = (p: SearchParams, key: 'destination' | 'themes', value: string): SearchParams => ({ ...p, [key]: Array.from(new Set([...(p[key] ?? []), value])) });
export function withoutParam(p: SearchParams, key: keyof SearchParams, value?: string): SearchParams {
  const next = { ...p } as Record<string, unknown>;
  if (value !== undefined && Array.isArray(next[key])) { const arr = (next[key] as string[]).filter((x) => x !== value); if (arr.length) next[key] = arr; else delete next[key]; }
  else delete next[key];
  return searchParamsSchema.parse(next);
}
```

- [ ] **Step 4: Itinerary with the route-draw motion (the page's signature)**

`Itinerary.tsx` (client):
```tsx
'use client';
import { useGSAP } from '@gsap/react';
import gsap from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';
import { useRef } from 'react';
import type { ItineraryDayView } from '@tripsmith/shared';
import { Icon } from '@/components/site/Icon';
gsap.registerPlugin(useGSAP, ScrollTrigger);
const meals = (m: ItineraryDayView['meals']) => [m.b && 'Breakfast', m.l && 'Lunch', m.d && 'Dinner'].filter(Boolean).join(' · ') || 'No meals';
export function Itinerary({ days }: { days: ItineraryDayView[] }) {
  const ref = useRef<HTMLDivElement>(null);
  useGSAP(() => {
    const mm = gsap.matchMedia();
    mm.add('(prefers-reduced-motion: no-preference)', () => {
      // the dashed route (the brand mark's path) draws down the timeline as you scroll; each day's stop lights up when the line reaches it
      gsap.fromTo('[data-route]', { scaleY: 0 }, { scaleY: 1, ease: 'none', scrollTrigger: { trigger: ref.current, start: 'top 70%', end: 'bottom 60%', scrub: 0.4 } });
      gsap.utils.toArray<HTMLElement>('[data-stop]').forEach((el) => gsap.fromTo(el, { backgroundColor: '#E3E8EC', color: '#5E6B76' }, { backgroundColor: '#1B4FD8', color: '#fff', duration: 0.3, scrollTrigger: { trigger: el, start: 'top 65%', toggleActions: 'play none none reverse' } }));
    });
    return () => mm.revert();
  }, { scope: ref });
  return (
    <div ref={ref} className="relative grid">
      <span aria-hidden className="absolute left-[16px] top-5 bottom-5 w-0.5 origin-top border-l-2 border-dashed border-pri" data-route />
      {days.map((d) => (
        <div key={d.dayNo} className="relative border-t border-line py-5 pl-[46px] last:border-b">
          <span data-stop className="absolute left-0 top-5 grid size-[34px] place-items-center rounded-[10px] bg-pri text-sm font-extrabold text-white">{d.dayNo}</span>
          <h3 className="mb-1.5 text-[19px]">{d.title}</h3>
          <p className="mb-2.5 max-w-[62ch] text-[15px] text-ink2">{d.description}</p>
          <div className="flex flex-wrap gap-2">
            <span className="inline-flex items-center gap-1.5 rounded-chip bg-bg2 px-2.5 py-1 text-xs font-bold text-mute"><Icon name="meal" className="size-[13px] text-pri" />{meals(d.meals)}</span>
            {d.stay && <span className="inline-flex items-center gap-1.5 rounded-chip bg-bg2 px-2.5 py-1 text-xs font-bold text-mute"><Icon name="bed" className="size-[13px] text-pri" />Stay · {d.stay}</span>}
          </div>
        </div>
      ))}
    </div>
  );
}
```
Under reduced motion the route line is fully drawn and every stop is lit (the `fromTo` never runs) — the page is complete at rest.
